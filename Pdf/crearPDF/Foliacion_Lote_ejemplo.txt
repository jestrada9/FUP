var documentos = [
    { area: "2310", num_docto: "1812004721", num_oficio: "1231004721" },
    { area: "2313", num_docto: "2418318864", num_oficio: "1231300412" }
];

$.ajax({
    url:  "../Pdf/crearPDF/Foliacion_Lote.php",
    type: "POST",
    dataType: "json",
    data: {
        documentos: window.btoa(JSON.stringify(documentos)),
        usu: usuario
    },
    success: function (r) {

        console.log("Total:", r.total,
                    "Foliados:", r.foliados,
                    "Omitidos:", r.omitidos,
                    "Errores:", r.errores,
                    "Mensaje:", r.mensaje);

        var txt = "";
        $.each(r.detalle, function (i, d) {
            txt += d.num_docto + "  ->  " +
                   (d.folio || "(sin folio)") +
                   "  [" + d.estado + "] " + d.mensaje + "\n";
        });
        $('#nd_aut').val(txt);

        // Sólo estos pasan a generar PDF
        var ok = $.grep(r.detalle, function (d) {
            return d.estado === "FOLIADO" || d.estado === "YA_FOLIADO";
        });
        console.log("Listos para PDF:", ok);
    },
    error: function (xhr) {
        console.log("Falló la petición:", xhr.status, xhr.responseText);
    }
});

/* =====================================================================
   QUÉ ESPERAR
   =====================================================================
   FOLIADO     el documento terminaba en 000001 y recibió folio nuevo.
   YA_FOLIADO  no terminaba en 000001; conserva su número, no se tocó.
   ERROR       no existe en rel_doctos, hay más de una fila con esa
               clave, o falló la transacción.

   Los folios de un mismo lote salen CONSECUTIVOS, en el orden en que
   mandaste los documentos.

   Si un documento falla dentro de la transacción, se revierte el LOTE
   COMPLETO y todos regresan como ERROR: no quedan folios a medias.
